// Biçim: #,0.0
[Yatırım (Milyon $)] =
    DIVIDE(SUM(FactYatirim[YatirimUSD]), 1000000)

// Biçim: #,0.0
[Toplam Yatırım (Milyar $)] =
    DIVIDE(SUM(FactYatirim[YatirimUSD]), 1000000000)

// Biçim: 0
[Yatırım Alan Ülke Sayısı] =
    DISTINCTCOUNT(FactYatirim[ISO3])

// Biçim: 0%
[Yatırım Kaydı Olan Ülke Oranı] =
    DIVIDE([Yatırım Alan Ülke Sayısı], DISTINCTCOUNT(DimUlke[ISO3]))

// Biçim: 0.0%
[ABD Payı %] =
    DIVIDE(CALCULATE(SUM(FactYatirim[YatirimUSD]), REMOVEFILTERS(DimUlke), DimUlke[ISO3] = "USA"), CALCULATE(SUM(FactYatirim[YatirimUSD]), REMOVEFILTERS(DimUlke)))

// Biçim: 0.0%
[İlk 10 Ülke Payı %] =
    VAR Ilk10 = TOPN(10, ALLSELECTED(DimUlke[UlkeTR]), [Yatırım (Milyon $)], DESC)
    RETURN DIVIDE(SUMX(Ilk10, [Yatırım (Milyon $)]), CALCULATE([Yatırım (Milyon $)], ALLSELECTED(DimUlke)))

// Biçim: #,0
[Top 10 Yatırım (Milyon $)] =
    VAR Sira = RANKX(ALLSELECTED(DimUlke[UlkeTR]), [Yatırım (Milyon $)], , DESC, Dense)
    RETURN IF(Sira <= 10 && [Yatırım (Milyon $)] > 0, [Yatırım (Milyon $)])

// Biçim: #,0
[Büyük Model Sayısı] =
    VAR SonYil = MAX(DimYil[Yil])
    VAR Deger = CALCULATE(SUM(FactModel[BuyukModelKumulatif]), DimYil[Yil] = SonYil)
    RETURN IF(Deger > 0, Deger)

// Biçim: 0
[Büyük Model Üreten Ülke Sayısı] =
    VAR SonYil = MAX(DimYil[Yil])
    RETURN COUNTROWS(FILTER(VALUES(DimUlke[ISO3]), CALCULATE(SUM(FactModel[BuyukModelKumulatif]), DimYil[Yil] = SonYil) > 0))

// Biçim: #,0.0
[Türkiye Yatırım (Milyon $)] =
    CALCULATE([Yatırım (Milyon $)], REMOVEFILTERS(DimUlke), DimUlke[ISO3] = "TUR")

[Kayıtlı Yıl Nüfusu] =
    SUMX(CROSSJOIN(VALUES(DimUlke[ISO3]), VALUES(DimYil[Yil])), IF(CALCULATE(SUM(FactYatirim[YatirimUSD])) > 0, CALCULATE(SUM(FactKalkinma[Nufus]))))

// Biçim: #,0.0
[Nüfus (Milyon, son yıl)] =
    VAR SonYil = MAX(DimYil[Yil])
    RETURN DIVIDE(CALCULATE(SUM(FactKalkinma[Nufus]), DimYil[Yil] = SonYil), 1000000)

// Biçim: #,0.00
[Kişi Başı Yatırım ($)] =
    VAR Gun = CROSSJOIN(VALUES(DimUlke[ISO3]), VALUES(DimYil[Yil]))
    VAR Pay = SUMX(Gun, IF(CALCULATE(SUM(FactKalkinma[Nufus])) > 0, CALCULATE(SUM(FactYatirim[YatirimUSD]))))
    RETURN DIVIDE(Pay, [Kayıtlı Yıl Nüfusu])

// Biçim: #,0.0
[Top 10 Kişi Başı Yatırım ($)] =
    VAR Havuz = FILTER(ALLSELECTED(DimUlke[UlkeTR]), [Nüfus (Milyon, son yıl)] >= 1)
    VAR Sira = RANKX(Havuz, [Kişi Başı Yatırım ($)], , DESC, Dense)
    RETURN IF([Nüfus (Milyon, son yıl)] >= 1 && Sira <= 10 && [Kişi Başı Yatırım ($)] > 0, [Kişi Başı Yatırım ($)])
